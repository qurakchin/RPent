# Overrides for the single-environment RoboDojo simulator stack.
#
# rlinf-robodojo-runtime pins Isaac Sim 5.1, the IsaacLab fork RoboDojo tracks,
# and the runtime minimums those two bring. Eight of those pins disagree with the
# agent stack or with rpent-openpi, so a plain resolve either fails or installs a
# combination that breaks at import time:
#
#   starlette==0.49.1 (isaaclab)      vs fastapi 0.115.7 needing <0.46
#   typing-extensions==4.12.2         vs tyro needing >=4.13, pydantic >=4.16
#   uvicorn==0.29.0 (isaacsim-kernel) vs mcp needing >=0.31.1
#   wrapt==1.16.0 (isaacsim-kernel)   vs swanlab needing >=1.17.0
#   filelock==3.13.1 (isaacsim-core)  vs rpent-openpi needing >=3.16.1
#   torch==2.7.1 (rpent-openpi)       vs the simulator requiring 2.7.0
#   pyperclip==1.8.0 (isaacsim-core)  vs pydantic-ai-slim[cli] needing >=1.9.0
#   idna==3.10 (isaacsim-kernel)      vs pydantic-ai-slim needing >=3.18
#   packaging==23.0 (isaacsim-core)   vs openai-codex needing >=26.2
#
# The robodojo component declares this file as an override, so
# `rpent-install install robot-robodojo` applies it automatically and the rest of
# repository keeps resolving the versions the other robots were validated
# against. Resolving these conflicts does not establish runtime compatibility.
starlette==0.45.3
typing-extensions>=4.16.0
uvicorn>=0.31.1
wrapt>=1.17.0
filelock>=3.16.1
torch==2.7.0
pyperclip>=1.9.0
idna>=3.18
packaging>=26.2
# NVIDIA's Linux and Windows wheels for this release have different metadata.
# Resolve against the Linux wheel itself so Windows-only pywin32 stays out.
isaacsim-core @ https://pypi.nvidia.com/isaacsim-core/isaacsim_core-5.1.0.0-cp311-none-manylinux_2_35_x86_64.whl#sha256=cfdd5d82ce455d654c68d37ff4fdd062190e3414289beb6f9b876a10a851b92d ; sys_platform == "linux" and platform_machine == "x86_64"
# The planner's Google provider needs websockets >=13. Version 13 retains the
# legacy asyncio API used by the simulator; the separate sim worker keeps 12.
websockets>=13,<14
